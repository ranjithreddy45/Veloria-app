# Veloria Grand scheduled jobs on the VPS (crontab -e as the deploy user).
# Replaces the Vercel crons. CRON_SECRET must match .env.production.
# The GitHub Actions workflows (fast/hourly) keep working unchanged — they call
# the public domain, which simply points at this server after cutover.
CRON_SECRET=__paste_from_.env.production__
0 2 * * * curl -fsS -H "Authorization: Bearer $CRON_SECRET" http://127.0.0.1:3100/api/cron/daily    >> /var/log/veloria-cron.log 2>&1
0 3 * * * curl -fsS -H "Authorization: Bearer $CRON_SECRET" http://127.0.0.1:3100/api/cron/frequent >> /var/log/veloria-cron.log 2>&1
